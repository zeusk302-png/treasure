# ─────────────────────────────────────────────
# robots.txt — 검색엔진(크롤러)에게 주는 안내문
# 이 파일은 "사이트 루트"에 두어야 합니다.
#   예) https://내사이트.com/robots.txt 로 열려야 함
# (Vercel/정적 사이트라면 public 폴더에 두면 루트로 올라갑니다)
# ─────────────────────────────────────────────

# User-agent: 어떤 크롤러에게 하는 말인지. * 는 "모든 크롤러"라는 뜻
User-agent: *

# Allow: 봐도 되는 경로 (허용)
Allow: /

# Disallow: 보지 말라고 막는 경로 (차단)
# 아래는 "관리자 페이지와 임시 폴더만" 검색에서 빼는 예시입니다.
# (왜 폴더만 막나) 막는 경로는 반드시 "/admin/"처럼 구체적으로 적습니다.
#   여기에 "Disallow: /" (슬래시 하나)만 적으면 사이트 전체가 차단되어
#   검색 결과에서 통째로 사라지기 때문에, 운영 사이트에는 절대 그 줄을 두지 않습니다.
Disallow: /admin/
Disallow: /private/
Disallow: /tmp/

# Sitemap: 사이트 지도(있다면) 위치를 알려 줍니다. 없으면 이 줄은 지워도 됩니다.
Sitemap: https://내사이트.com/sitemap.xml
